# Requirements for the Ethos Manual Translator
# Install with: pip install -r requirements_translator.txt
#
# The GUI itself is tkinter (Python stdlib, nothing to install).

# Talking to the GitHub API (auth check, branches, contents, PRs)
PyGithub>=2.3

# PyGithub pulls in pyjwt[crypto] (for GitHub App JWT auth, which this app
# never uses -- we only do PAT auth), which pulls in `cryptography`.
# cryptography 49.0.0+ stopped publishing wheels for 32-bit Windows, so an
# unpinned install resolves the latest version, finds no win32 wheel, and
# falls back to building from source -- which fails on the win32 CI runner
# (no Rust toolchain). Pinned below to the last version with a win32 wheel.
cryptography<49

# Storing the user's PAT in the OS credential store (Windows Credential
# Manager / macOS Keychain / Linux Secret Service) instead of a plain file
keyring>=25.0

# Parsing mkdocs.yml (nav:, plugins.i18n.languages) fetched over the API
PyYAML>=6.0

# Rendering the "Preview" button's HTML using the same extensions
# ethos-manual-rework's own mkdocs.yml markdown_extensions: list does
Markdown>=3.5
pymdown-extensions>=10.0
